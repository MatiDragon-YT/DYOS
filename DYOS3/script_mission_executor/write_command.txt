    then
        FILE_OFFSET = 0
        0AC8: SMX_BUFFER_MAIN = allocate_memory_size FILE_SIZE

        while FILE_OFFSET < FILE_SIZE
            0AD5: file_seek FILE_HANDLE offset FILE_OFFSET origin 0
            0A9D: read_file FILE_HANDLE size 1 to 0@

            2402: write_memory_with_offset SMX_BUFFER_MAIN offset FILE_OFFSET size 1 value 0@
            FILE_OFFSET += 1
        end

        SMX_CURRENT_OFFSET = 0
        2401: read_memory_with_offset SMX_BUFFER_MAIN offset SMX_CURRENT_OFFSET size 4 store_to SMX_VERSION_IN_EDIT

        0AD5: file_seek FILE_HANDLE offset FILE_SIZE origin 0

        0AC9: free_allocated_memory SMX_BUFFER_MAIN
        FILE_OFFSET = 0
    end
    if SMX_VERSION_IN_EDIT == 0
    then
        printf "No se pudo detectar la version del SMX." WAIT_ERROR
        call @waitPlis 1 WAIT_ERROR
        return
    else
        if SMX_VERSION_IN_EDIT < DYOS_CODE_VERSION
        then printf "Actualiza el SMX para poder editarlo, de...~n~SMX %i a %i" WAIT_ERROR SMX_VERSION_IN_EDIT DYOS_CODE_VERSION
            call @waitPlis 1 WAIT_ERROR
            return
        end
        
        if SMX_VERSION_IN_EDIT > DYOS_CODE_VERSION
        then
            printf "Actualiza el DYOS para poder editarlo, de...~n~DYOS %i a %i" WAIT_ERROR SMX_VERSION_IN_EDIT DYOS_CODE_VERSION
            call @waitPlis 1 WAIT_ERROR
            return
        end
    end

    if FILE_COMMAND >= 0x8000
    then
        if SMX_ENTER_CONDITION == false
        then
            SMX_ENTER_CONDITION = true

        end

        SMX_ENTER_CONDITION = false
    end
// END_AREA

else

        end

    end
