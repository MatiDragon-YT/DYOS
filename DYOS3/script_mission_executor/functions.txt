:File_Chech_Exist
    if CURRENT_PLATFORM == __PLATFORM_WINDOWS
    then DIR_DYOS = "CLEO/0.smx"
    else DIR_DYOS = "/0.smx"
    end
    if 0AAB:  does_file_exist DIR_DYOS
    then
        FILE_EXIST = true
        
        0A9A: FILE_HANDLE = DIR_DYOS "rb"
        0A9C: FILE_SIZE = file FILE_HANDLE size
        0A9B: file_close FILE_HANDLE
        
        SMX_ENTER_CONDITION = false
        SMX_COMMAND_IS_CONDITIONAL = 0
        SMX_COND_START_OFFSET = FILE_SIZE
        SMX_IN_CONDITION_BLOCK = false
    else
        if 0A9A: FILE_HANDLE = DIR_DYOS "rb"
        then
            Fs.Close(FILE_HANDLE)
            FILE_EXIST = true
        else
            Text.PrintHelp('M00001?') // ~p~SMX:~s~~n~~t~Creando archivo "./0.smx".

            if 0A9A: FILE_HANDLE = DIR_DYOS "w+b"
            then
                call @waitPlis 1 150
                0A9E: write_file FILE_HANDLE size 4 from 0 // 300010
                call @waitPlis 1 150
                Fs.Close(FILE_HANDLE)
                call @waitPlis 1 150

                FILE_EXIST = true

                if 0AAB:  does_file_exist DIR_DYOS
                then
                    reset_file()
                    Text.ClearHelp()
                    Text.PrintHelp('M00001_') // ~p~SMX:~s~~n~~t~Archivo creado.
                    goto @File_Chech_Exist
                end
            end

            FILE_EXIST = false
            Text.PrintHelp('M00000+') // ~p~SMX:~s~~n~~t~Error con "./0.smx".
            FILE_HANDLE = 0
        end
    end

return

    SMX_CURRENT_OFFSET += 1

    2000: 29@ = get_cleo_arg_count

    0A9E: write_file FILE_HANDLE size 4 to 0@

        0A9E: write_file FILE_HANDLE size 1 to 0

        0A9E: write_file FILE_HANDLE size 1 to 0

    if 29@ > 4

        0A9E: write_file FILE_HANDLE size 1 to 0

    if 29@ > 5
    then
        0A9E: write_file FILE_HANDLE size 1 to 0
        0A9E: write_file FILE_HANDLE size 4 to 5@

        0A9E: write_file FILE_HANDLE size 1 to 0

ret 0

:SMX_read_int_v2
    fixMagicPointer()

    if FILE_VERSION > 300015
    then
        // TYPE BYTES -> 
        2401: read_memory_with_offset SMX_BUFFER_MAIN offset SMX_CURRENT_OFFSET size 1 store_to 0@ // type
        SMX_CURRENT_OFFSET += 1

        // VALUE SAVE ->
        2401: read_memory_with_offset SMX_BUFFER_MAIN offset SMX_CURRENT_OFFSET size 4 store_to 1@ // value
        SMX_CURRENT_OFFSET += 4

        if 0@ == 1 // 1 = VARIABLE || 0 = CONSTANT
        then
            2@ = 1@ * 4 // offset
            2401: read_memory_with_offset SMX_BUFFER_VARIABLES offset 2@ size 4 store_to 1@
        end
    else
        2401: read_memory_with_offset SMX_BUFFER_MAIN offset SMX_CURRENT_OFFSET size 4 store_to 1@
        SMX_CURRENT_OFFSET += 4
    end
ret 1 1@

:SMX_read_float_v2
    fixMagicPointer()
    
    if FILE_VERSION > 300015
    then
        // TYPE BYTES -> 
        2401: read_memory_with_offset SMX_BUFFER_MAIN offset SMX_CURRENT_OFFSET size 1 store_to 0@
        SMX_CURRENT_OFFSET += 1

        if 0@ == 1 // 1 = VARIABLE || 0 = CONSTANT
        then
            // VALUE SAVE ->
            2401: read_memory_with_offset SMX_BUFFER_MAIN offset SMX_CURRENT_OFFSET size 4 store_to 1@
            SMX_CURRENT_OFFSET += 4

            2@ = 1@ * 4 // offset
            2401: read_memory_with_offset SMX_BUFFER_VARIABLES offset 2@ size 4 store_to 3@
        else
            // VALUE SAVE ->
            2401: read_memory_with_offset SMX_BUFFER_MAIN offset SMX_CURRENT_OFFSET size 4 store_to 3@
            SMX_CURRENT_OFFSET += 4
        end
    else
        2401: read_memory_with_offset SMX_BUFFER_MAIN offset SMX_CURRENT_OFFSET size 4 store_to 3@
        SMX_CURRENT_OFFSET += 4
    end
ret 1 3@

:reset_file
    if 0A9A: FILE_HANDLE = DIR_DYOS "wb"
    then
        FILE_OFFSET = 0
        
        // FILE_VERSION [XXX][XX]
        //
        // [XXX] : INT version
        // [XX] : INT build
        0A9E: write_file FILE_HANDLE size 4 from DYOS_CODE_VERSION
        SMX_VERSION_IN_EDIT = DYOS_CODE_VERSION

        FILE_COMMAND = VM_NOP
        SMX_write_command()
        Fs.Close(FILE_HANDLE)
    end
return

:clear_buffer
    // 0@ buffer_id

    int 0@
    
    fixMagicPointer()
    if 0@ == BUFFER_ACTORS
    then
        while SMX_NUM_ACTORS > 0
            call @SMX_Array_Actor_Get 2 SMX_NUM_ACTORS B_ACTOR_HANDLE result SMX_ACTOR

            if Actor.Defined(SMX_ACTOR)
            then Actor.Remove(SMX_ACTOR)
            end

            call @SMX_Array_Actor_Set 3 SMX_NUM_ACTORS B_ACTOR_HANDLE value 0

            SMX_NUM_ACTORS -= 1
        end
    end

    if 0@ == BUFFER_TARGET_BLIPS
    then
        while SMX_NUM_TARGET_BLIPS > 0

            {
            if and
                CURRENT_PLATFORM == __PLATFORM_ANDROID
                bit_is_true DYOS_FLAGS DEBUG_MODE
            then
                printf "SMX_BUFFER_TARGET_BLIPS %i~n~SMX_NUM_TARGET_BLIPS %i" WAIT_DEBUG SMX_BUFFER_TARGET_BLIPS SMX_NUM_TARGET_BLIPS
                call @waitPlis 1 WAIT_DEBUG
            end
            }

            // ERROR AT ANDROID &268(30@,1i) = &400(30@,1i) * 4
            // VERY DESMONTED
            SMX_TEMP_OFFSET = SMX_NUM_TARGET_BLIPS
            SMX_TEMP_OFFSET *= 4
            
            2401: read_memory_with_offset SMX_BUFFER_TARGET_BLIPS offset SMX_TEMP_OFFSET size 4 store_to SMX_TARGET_BLIP

            if bit_is_true DYOS_FLAGS DEBUG_MODE
            then
                printf "index %i offset %i handle %i" WAIT_DEBUG params SMX_NUM_TARGET_BLIPS SMX_TEMP_OFFSET SMX_TARGET_BLIP
                call @waitPlis 1 WAIT_DEBUG
            end

            if Blip.Defined(SMX_TARGET_BLIP)
            then Blip.Remove(SMX_TARGET_BLIP)
            end

            2402: write_memory_with_offset SMX_BUFFER_TARGET_BLIPS offset SMX_TEMP_OFFSET size 4 value 0

            SMX_NUM_TARGET_BLIPS -= 1
        end
    end

    if 0@ == BUFFER_TARGET_SPHERES
    then
        while SMX_NUM_TARGET_SPHERES > 0
            SMX_TEMP_OFFSET = SMX_NUM_TARGET_SPHERES
            SMX_TEMP_OFFSET *= 4
            2401: read_memory_with_offset SMX_BUFFER_TARGET_SPHERES offset SMX_TEMP_OFFSET size 4 store_to SMX_TARGET_SPHERE

            if SMX_TARGET_SPHERE <> 0
            then Sphere.Remove(SMX_TARGET_SPHERE)
            end

            2402: write_memory_with_offset SMX_BUFFER_TARGET_SPHERES offset SMX_TEMP_OFFSET size 4 value 0

            SMX_NUM_TARGET_SPHERES -= 1
        end
    end

    :exit_clear_buffer
ret 0
