Text.ClearHelp()
Menu_Display_Hide()
Menu_Take_Selection()
UnAttachPlayer()

VELOCITY_MOVEMENT_FREE = 0.3

repeat
  if Menu_Close_Forced()
  then jump @Menu_Close
  end
	SMX_Xf = 0.0
	SMX_Yf = 0.0
	SMX_Zf = -1.0

	if VK_No()
	then VELOCITY_MOVEMENT_FREE -= 0.02
	end
	if VK_Yes()
	then VELOCITY_MOVEMENT_FREE += 0.02
	end

	if VELOCITY_MOVEMENT_FREE < 0.0
	then VELOCITY_MOVEMENT_FREE = 0.0
	end
	if VELOCITY_MOVEMENT_FREE > 2.0
	then VELOCITY_MOVEMENT_FREE = 2.0
	end


	if bit_is_false DYOS_FLAGS CAMARA_CINEMATICA
	then
		VELOCITY_MOVEMENT_CALCULATE = VELOCITY_MOVEMENT_FREE
		VELOCITY_MOVEMENT_CALCULATE *= 10000.0

		printf ">>> %.2f >>>" 10 VELOCITY_MOVEMENT_FREE

		call @ShowData 0
	end
	

	SMX_Af = 0.0
	0078: SMX_Af += frame_delta_time * VELOCITY_MOVEMENT_FREE // (float)

	Player_Memoize_angle = Actor.GetAngle(hPlayer) 
	
	if VK_Right()
	then SMX_Xf += SMX_Af
	end
	if VK_Left()
	then SMX_Xf -= SMX_Af
	end  
	if ParaAdelante()
	then SMX_Yf += SMX_Af
	end
	if ParaAtras()
	then SMX_Yf -= SMX_Af
	end 
	if VK_Up()
	then SMX_Zf += SMX_Af
	end 
	if VK_Down()
	then SMX_Zf -= SMX_Af
	end

	SMX_Af *= 5.0
	if VK_Plus() 
	then Player_Memoize_angle -= SMX_Af
	end
	if VK_Minus()
	then Player_Memoize_angle += SMX_Af
	end 

	Actor.GetPositionAndOffset(Player_Memoize_pos_x, Player_Memoize_pos_y, Player_Memoize_pos_z, hPlayer, SMX_Xf, SMX_Yf, SMX_Zf)
	Actor.SetPosition(hPlayer, Player_Memoize_pos_x, Player_Memoize_pos_y, Player_Memoize_pos_z)
	Actor.SetAngle(hPlayer) = Player_Memoize_angle
//	GirarCamara()




	if VK_Enter()
	then break
	end
until Menu_Return()
Menu_Display_Show()
